func Split(src []byte) ([]Chunk, error) {
    fset := token.NewFileSet()
    f, err := parser.ParseFile(fset, "", src, parser.ParseComments)
    if err != nil {
        return nil, err
    }
    var out []Chunk
    for _, d := range f.Decls {
        a, b := fset.Position(d.Pos()), fset.Position(d.End())
        out = append(out, Chunk{Text: src[a.Offset:b.Offset], Line: a.Line})
    }
    return out, nil
}
